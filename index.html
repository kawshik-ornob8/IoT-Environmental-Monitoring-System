Firmware Logic

Connectivity & Logic

The ESP32 reads 7 sensors and uploads the payload to ThingSpeak every 5 seconds.

// 1. Initialization
void setup() {
  WiFi.begin(ssid, password);
  ThingSpeak.begin(client);
  pinMode(buzzer_pin, OUTPUT);
}

// 2. Core Loop
void loop() {
  // Read Sensors
  int waterLevel = analogRead(35);
  float temp = dht.readTemperature();
  
  // Safety Logic: Low Water Alert
  if (waterLevel < 100) {
    digitalWrite(buzzer_pin, HIGH);
  } else {
    digitalWrite(buzzer_pin, LOW);
  }

  // Cloud Upload
  ThingSpeak.setField(1, waterLevel);
  ThingSpeak.setField(2, mq7Value);
  ThingSpeak.setField(5, temp);
  
  ThingSpeak.writeFields(channelID, apiKey);
  delay(5000);
}
